  get_target_property(production_sources ${target} SOURCES)
  foreach(source IN LISTS production_sources)
    get_source_file_property(source_language "${source}" TARGET_DIRECTORY ${target} LANGUAGE)
    if(source_language STREQUAL "NOTFOUND")
      get_target_property(target_source_directory ${target} SOURCE_DIR)
      get_filename_component(absolute_source "${source}" ABSOLUTE BASE_DIR "${target_source_directory}")
      set(package_source_root "${CMAKE_SOURCE_DIR}/src")
      cmake_path(IS_PREFIX package_source_root "${absolute_source}" NORMALIZE package_source)
      if(package_source)
        get_filename_component(source_directory "${absolute_source}" DIRECTORY)
        get_source_file_property(source_language "${absolute_source}" DIRECTORY "${source_directory}" LANGUAGE)
      endif()
      if(source_language STREQUAL "NOTFOUND")
        get_filename_component(source_extension "${absolute_source}" LAST_EXT)
        string(SUBSTRING "${source_extension}" 1 -1 source_extension)
        if(source_extension IN_LIST CMAKE_CXX_SOURCE_FILE_EXTENSIONS)
          set(source_language CXX)
        endif()
      endif()
    endif()
    if(NOT source_language STREQUAL "CXX")
      message(FATAL_ERROR "Own production source must compile as C++26: ${source}")
    endif()
  endforeach()
endforeach()
